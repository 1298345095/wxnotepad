cmake_minimum_required(VERSION 3.25)
project(worker_measurement_prototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.3.3 EXACT CONFIG REQUIRED)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
get_filename_component(default_port "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PORT_ROOT "${default_port}" CACHE PATH "Prototype checkout")
set(PORT_BUILD "" CACHE PATH "Existing baseline Release build, from 81515bf or its code-equivalent 05b7d86")
set(CORE_ROOT "" CACHE PATH "Pinned prepared original-core source")
if(NOT PORT_BUILD OR NOT EXISTS "${CORE_ROOT}/wxnotepad-source.json")
    message(FATAL_ERROR "Set PORT_BUILD to the baseline build and CORE_ROOT to its prepared original-core directory")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
    "${PORT_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(core_probe "${CMAKE_CURRENT_BINARY_DIR}/editor_core_probe.cpp")
if(WIN32)
    set(measurement "${PORT_ROOT}/src/editor/platform/text_measurement_msw.cpp")
else()
    set(measurement "${PORT_ROOT}/src/editor/platform/text_measurement_gtk.cpp")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-egl wayland-cursor)
endif()
add_library(worker_measurement OBJECT "${measurement}")
add_executable(worker_metrics WIN32 "${PORT_ROOT}/tests/measurement_threads.cpp" $<TARGET_OBJECTS:worker_measurement>)
add_executable(worker_resize WIN32 "${PORT_ROOT}/tests/wrap_resize.cpp" $<TARGET_OBJECTS:worker_measurement>)
add_executable(worker_fonts WIN32 "${PORT_ROOT}/tests/font_smoke.cpp" $<TARGET_OBJECTS:worker_measurement>)
add_executable(worker_app WIN32 "${PORT_ROOT}/src/app/main.cpp" $<TARGET_OBJECTS:worker_measurement>)
add_executable(worker_layout WIN32 benchmark_probe.cpp "${core_probe}" $<TARGET_OBJECTS:worker_measurement>)
add_executable(worker_baseline_layout WIN32 benchmark_probe.cpp "${core_probe}")
add_executable(worker_fixed_layout WIN32 "${CMAKE_CURRENT_BINARY_DIR}/benchmark_fixed.cpp" "${core_probe}" $<TARGET_OBJECTS:worker_measurement>)
add_executable(baseline_fixed_layout WIN32 "${CMAKE_CURRENT_BINARY_DIR}/benchmark_fixed.cpp" "${core_probe}")
foreach(target worker_measurement worker_metrics worker_resize worker_fonts worker_app worker_layout worker_baseline_layout worker_fixed_layout baseline_fixed_layout)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PORT_ROOT}/src" "${PORT_ROOT}/tests"
        "${PORT_ROOT}/src/editor/platform" "${CORE_ROOT}/scintilla/include" "${CORE_ROOT}/scintilla/src"
        "${CORE_ROOT}/scintilla/lexlib" "${CORE_ROOT}/scintilla/win32")
    target_compile_definitions(${target} PRIVATE __WX__ BOOST_REGEX_STANDALONE NO_CXX11_REGEX NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
        target_compile_options(${target} PRIVATE /utf-8 /Zi)
        if(NOT target STREQUAL worker_measurement)
            target_sources(${target} PRIVATE "${PORT_ROOT}/src/app/windows.rc")
            target_link_options(${target} PRIVATE /MANIFEST:NO /DEBUG)
            target_link_libraries(${target} PRIVATE "${PORT_BUILD}/app_shell.lib" "${PORT_BUILD}/editor_features.lib"
                "${PORT_BUILD}/editor_runtime.lib" imm32 shlwapi ole32 gdi32 advapi32 shell32)
        endif()
    else()
        if(NOT target STREQUAL worker_measurement)
            target_link_libraries(${target} PRIVATE "${PORT_BUILD}/libapp_shell.a" "${PORT_BUILD}/libeditor_features.a" "${PORT_BUILD}/libeditor_runtime.a")
        endif()
    endif()
    target_link_libraries(${target} PRIVATE wx::core wx::base)
    if(NOT WIN32)
        target_link_libraries(${target} PRIVATE PkgConfig::GTK PkgConfig::WAYLAND)
    endif()
endforeach()
if(MSVC)
    target_compile_options(worker_measurement PRIVATE /GR- /Zc:char8_t-)
    set_source_files_properties("${core_probe}" PROPERTIES COMPILE_OPTIONS "/GR-;/Zc:char8_t-")
else()
    target_compile_options(worker_measurement PRIVATE -fno-char8_t)
    set_source_files_properties("${core_probe}" PROPERTIES COMPILE_OPTIONS "-fno-char8_t")
endif()
enable_testing()
add_test(NAME worker_parallel_metrics COMMAND worker_metrics)
add_test(NAME worker_wrapped_resize COMMAND worker_resize)
add_test(NAME worker_font_metrics COMMAND worker_fonts)
set_tests_properties(worker_parallel_metrics PROPERTIES TIMEOUT 90)
set_tests_properties(worker_wrapped_resize PROPERTIES TIMEOUT 120)
set_tests_properties(worker_font_metrics PROPERTIES TIMEOUT 20)
