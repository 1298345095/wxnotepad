cmake_minimum_required(VERSION 3.25)
project(directwrite_prototype LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "The DirectWrite experiment is Windows-only; Linux retains Pango/Cairo")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.3.3 EXACT CONFIG REQUIRED)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(PORT_BUILD "" CACHE PATH "Existing Release baseline libraries (81515bf/05b7d86)")
set(CORE_ROOT "" CACHE PATH "Pinned prepared original core")
if(NOT PORT_BUILD OR NOT EXISTS "${CORE_ROOT}/wxnotepad-source.json")
    message(FATAL_ERROR "Set PORT_BUILD and CORE_ROOT to the existing baseline build/prepared core")
endif()
file(READ "${CORE_ROOT}/wxnotepad-source.json" source_record)
string(JSON core_revision GET "${source_record}" revision)
if(NOT core_revision STREQUAL "89b8b36e0ee2c23f73be922315f17261ccfbbe1a")
    message(FATAL_ERROR "The DirectWrite experiment requires the project's pinned original core")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
    "${CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set(profile_root "${CMAKE_CURRENT_SOURCE_DIR}/../prototype_worker_measurement")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${profile_root}/prepare_sources.py"
    "${PORT_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_library(directwrite_platform OBJECT directwrite_surface.cpp "${PORT_ROOT}/src/editor/platform/wx_platform.cpp")
add_library(directwrite_host OBJECT "${PORT_ROOT}/src/editor/editor_core.cpp")
add_library(directwrite_profile OBJECT "${CMAKE_CURRENT_BINARY_DIR}/editor_core_probe.cpp")
add_executable(directwrite_app WIN32 "${PORT_ROOT}/src/app/main.cpp"
    $<TARGET_OBJECTS:directwrite_platform> $<TARGET_OBJECTS:directwrite_host>)
add_executable(directwrite_resize WIN32 "${PORT_ROOT}/tests/wrap_resize.cpp"
    $<TARGET_OBJECTS:directwrite_platform> $<TARGET_OBJECTS:directwrite_host>)
add_executable(directwrite_oracle WIN32 native_oracle.cpp
    $<TARGET_OBJECTS:directwrite_platform> $<TARGET_OBJECTS:directwrite_host>)
add_executable(directwrite_layout WIN32 "${CMAKE_CURRENT_BINARY_DIR}/benchmark_fixed.cpp"
    $<TARGET_OBJECTS:directwrite_platform> $<TARGET_OBJECTS:directwrite_profile>)
foreach(probe caret_appearance auto_insertion automatic_completion completion_popup)
    add_executable(directwrite_${probe} WIN32 "${PORT_ROOT}/tests/${probe}.cpp"
        $<TARGET_OBJECTS:directwrite_platform> $<TARGET_OBJECTS:directwrite_host>)
endforeach()
foreach(target directwrite_platform directwrite_host directwrite_profile directwrite_app directwrite_resize directwrite_oracle directwrite_layout
        directwrite_caret_appearance directwrite_auto_insertion directwrite_automatic_completion directwrite_completion_popup)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${profile_root}" "${PORT_ROOT}/src"
        "${PORT_ROOT}/tests" "${PORT_ROOT}/src/editor/platform" "${CORE_ROOT}/scintilla/include"
        "${CORE_ROOT}/scintilla/src" "${CORE_ROOT}/scintilla/lexlib" "${CORE_ROOT}/scintilla/win32")
    target_compile_definitions(${target} PRIVATE WXNOTEPAD_PROTOTYPE_DIRECTWRITE __WX__ BOOST_REGEX_STANDALONE
        NO_CXX11_REGEX NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_compile_options(${target} PRIVATE /utf-8 /Zi)
    target_link_libraries(${target} PRIVATE wx::core wx::base)
    get_target_property(target_type ${target} TYPE)
    if(NOT target_type STREQUAL "OBJECT_LIBRARY")
        target_sources(${target} PRIVATE "${PORT_ROOT}/src/app/windows.rc")
        target_link_options(${target} PRIVATE /MANIFEST:NO /DEBUG)
        target_link_libraries(${target} PRIVATE "${PORT_BUILD}/app_shell.lib" "${PORT_BUILD}/editor_features.lib"
            "${PORT_BUILD}/editor_runtime.lib" imm32 shlwapi ole32 gdi32 advapi32 shell32)
    endif()
endforeach()
foreach(target directwrite_platform directwrite_host directwrite_profile)
    target_compile_options(${target} PRIVATE /GR- /Zc:char8_t-)
endforeach()
enable_testing()
add_test(NAME directwrite_wrapped_resize COMMAND directwrite_resize)
set_tests_properties(directwrite_wrapped_resize PROPERTIES TIMEOUT 120)
add_test(NAME directwrite_original_metrics_and_raster COMMAND directwrite_oracle)
set_tests_properties(directwrite_original_metrics_and_raster PROPERTIES TIMEOUT 90)
add_test(NAME directwrite_caret_appearance COMMAND directwrite_caret_appearance)
set_tests_properties(directwrite_caret_appearance PROPERTIES TIMEOUT 60)
add_test(NAME directwrite_caret_dpi96 COMMAND directwrite_caret_appearance)
set_tests_properties(directwrite_caret_dpi96 PROPERTIES TIMEOUT 60 ENVIRONMENT "WXNOTEPAD_CARET_DPI96=1")
add_test(NAME directwrite_auto_insertion COMMAND directwrite_auto_insertion)
set_tests_properties(directwrite_auto_insertion PROPERTIES TIMEOUT 90)
add_test(NAME directwrite_automatic_completion COMMAND directwrite_automatic_completion)
set_tests_properties(directwrite_automatic_completion PROPERTIES TIMEOUT 120)
add_test(NAME directwrite_completion_popup COMMAND directwrite_completion_popup)
set_tests_properties(directwrite_completion_popup PROPERTIES TIMEOUT 60)
